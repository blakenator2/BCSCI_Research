inp_img  = Input((WINDOW,WINDOW,2), dtype=tf.float32)
inp_feat = Input((X_feat.shape[1],), dtype=tf.float32)

def conv_block(x,f,dr):
    x=layers.Conv2D(f,3,padding="same")(x)
    x=layers.BatchNormalization()(x); x=layers.Activation("relu")(x)
    x=layers.Conv2D(f,3,padding="same")(x)
    x=layers.BatchNormalization()(x); x=layers.Activation("relu")(x)
    x=layers.MaxPool2D()(x)
    return layers.Dropout(dr)(x) if dr>0 else x

x = conv_block(inp_img,32,0.2)
x = conv_block(x,      64,0.3)
x = conv_block(x,     128,0.4)
x = layers.Flatten()(x)
m = layers.Concatenate()([x, inp_feat])
m = layers.Dense(256)(m); m=layers.BatchNormalization()(m)
m = layers.Activation("relu")(m); m=layers.Dropout(0.5)(m)
m = layers.Dense(128)(m); m=layers.BatchNormalization()(m)
m = layers.Activation("relu")(m); m=layers.Dropout(0.5)(m)
out=layers.Dense(1,activation="sigmoid",bias_initializer=bias_init)(m)

cnn = models.Model([inp_img,inp_feat], out)
cnn.compile(optimizer="adam", loss="binary_crossentropy", metrics=["accuracy"])
